    Bantuan Sintaks SQL

    CREATE DATABASE
    CREATE DATABASE inventaris_lab;
USE inventaris_lab;
    CREATE TABLE
    CREATE TABLE contoh (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nama VARCHAR(100) NOT NULL
);
    ALTER TABLE
    -- Satu ALTER, dua kolom
ALTER TABLE barang
ADD COLUMN harga DECIMAL(12,2),
ADD COLUMN merek VARCHAR(50);

-- Atau dua ALTER terpisah
ALTER TABLE barang ADD harga DECIMAL(12,2);
ALTER TABLE barang ADD merek VARCHAR(50);
    INSERT
    INSERT INTO contoh (nama)
VALUES ('Data A'), ('Data B');
    UPDATE
    UPDATE contoh
SET nama = 'Baru'
WHERE id = 1;

-- Operasi angka juga didukung
UPDATE barang
SET jumlah = jumlah - 3
WHERE kode_barang = 'BRG003';
    DELETE
    DELETE FROM contoh
WHERE id = 1;
    SELECT / ORDER BY
    SELECT * FROM contoh
WHERE nama = 'Data A'
ORDER BY nama ASC;
    JOIN
    SELECT a.nama, b.kategori
FROM barang a
JOIN kategori b
ON a.id_kategori = b.id_kategori;
    Melihat struktur
    SHOW TABLES;
DESCRIBE barang;
